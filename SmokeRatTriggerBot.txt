--[[ La ui ]]--
local screenGui, toggleBtn, keybindBtn, hideBtn, hideKeybindBtn, autoEquipBtn, destroyBtn


local function updateToggleVisuals()
	if toggleBtn then
		toggleBtn.Text = "SmokeGuiRatXD: " .. (enabled and "ON" or "OFF")
		local targetColor = enabled and Color3.fromRGB(35, 180, 100) or Color3.fromRGB(60, 60, 70)
		TweenService:Create(toggleBtn, TweenInfo.new(0.2), {BackgroundColor3 = targetColor}):Play()
	end
end

local function updateAutoEquipVisuals()
	if autoEquipBtn then
		autoEquipBtn.Text = "Auto Equip: " .. (autoEquip and "ON" or "OFF")
		local targetColor = autoEquip and Color3.fromRGB(35, 140, 180) or Color3.fromRGB(60, 60, 70)
		TweenService:Create(autoEquipBtn, TweenInfo.new(0.2), {BackgroundColor3 = targetColor}):Play()
	end
end


local isHidden = false
local function toggleHideGui()
	isHidden = not isHidden
	if toggleBtn then toggleBtn.Visible = not isHidden end
	if keybindBtn then keybindBtn.Visible = not isHidden end
	if hideBtn then hideBtn.Visible = not isHidden end
	if hideKeybindBtn then hideKeybindBtn.Visible = not isHidden end
	if autoEquipBtn then autoEquipBtn.Visible = not isHidden end
	if destroyBtn then destroyBtn.Visible = not isHidden end
end


UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if waitingForKey then
		if input.UserInputType == Enum.UserInputType.Keyboard then
			if waitingForKey == "toggle" then
				toggleKey = input.KeyCode
				if keybindBtn then
					keybindBtn.Text = "Key: " .. toggleKey.Name
					keybindBtn.BackgroundColor3 = Color3.fromRGB(50, 50, 60)
				end
			elseif waitingForKey == "hide" then
				hideKey = input.KeyCode
				if hideKeybindBtn then
					hideKeybindBtn.Text = "Key: " .. hideKey.Name
					hideKeybindBtn.BackgroundColor3 = Color3.fromRGB(50, 50, 60)
				end
			end
			waitingForKey = nil
		end
		return
	end

	if input.UserInputType == Enum.UserInputType.Keyboard then
		if input.KeyCode == toggleKey then
			enabled = not enabled
			updateToggleVisuals()
		elseif input.KeyCode == hideKey then
			toggleHideGui()
		end
	end
end)

RunService.RenderStepped:Connect(function()
	if enabled == false then return end
	if tick() - lastShot < fireCooldown then return end

	local target = mouse.Target
	if not target then return end
	if not target:IsDescendantOf(Workspace) then return end

	local character = target:FindFirstAncestorOfClass("Model")
	if not character then return end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid or humanoid.Health <= 0 then return end
	if character == localPlayer.Character then return end

	local char = localPlayer.Character
	if not char then return end

	local tool = char:FindFirstChild("Kawaii Revolver")
	
	-- Si el auto equip está activado y no tenemos el arma en la mano, la buscamos y equipamos
	if not tool and autoEquip then
		local backpack = localPlayer:FindFirstChild("Backpack")
		if backpack then
			tool = backpack:FindFirstChild("Kawaii Revolver")
			if tool then
				tool.Parent = char
				task.wait(0.05)
			end
		end
	end


	if not tool then return end

	lastShot = tick()

	pcall(function()
		local mousePos = UserInputService:GetMouseLocation()
		VirtualInputManager:SendMouseButtonEvent(mousePos.X, mousePos.Y, 0, true, game, 1)
		task.wait(0.02)
		VirtualInputManager:SendMouseButtonEvent(mousePos.X, mousePos.Y, 0, false, game, 1)
	end)
end)
